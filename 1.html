} else { appData = parsedData; appData.isFirstTime = false; // 确保成员有seasonExperience字段 appData.members.forEach(member => { if (member.seasonExperience === undefined) { member.seasonExperience = 0; } if (member.experience === undefined) { member.experience = 0; } // 检查是否有旧的快速操作数据（5和10的），如果是则更新为1和2 const oldQuickActions = [ { points: 5, label: "+5", type: "add" }, { points: 10, label: "+10", type: "add" }, { points: 5, label: "-5", type: "deduct" }, { points: 10, label: "-10", type: "deduct" } ];
